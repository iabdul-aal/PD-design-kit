\begin{frame}{Motivation and Targets}
\begin{itemize}
\item Intra-datacenter PAM-4 receivers need \textbf{high responsivity}, \textbf{very low dark current}, and enough bandwidth for \SI{53.125}{\giga\baud} PAM-4 lanes.
\item A \textbf{Ge-on-Si vertical n-i-p photodiode} offers a strong fit with standard SOI photonics.
\item The chapter benchmarks the design against Yang Shi et al., \textit{Photonics Research} 12(1), 2024.
\end{itemize}
\vspace{0.5em}
\centering
\begin{tabular}{ll}
\toprule
Metric & Target \\
\midrule
Responsivity & $\geq \SI{0.95}{\ampere\per\watt}$ \\
Dark current & $\leq \SI{1.3}{\nano\ampere}$ \\
Bandwidth & $\geq \SI{103}{\giga\hertz}$ \\
Detectivity & $\geq 2.95\times10^{10}$~Jones \\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}{Device Architecture}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}
\item Vertical \textbf{n-i-p} stack on a \textbf{\SI{220}{\nano\metre} SOI} platform.
\item \textbf{\SI{50}{\nano\metre} n++ Ge} top contact layer.
\item \textbf{\SI{350}{\nano\metre} intrinsic Ge} absorber.
\item \textbf{p++ Si} platform for the anode path.
\item \textbf{U-shaped electrode} reduces the RC penalty while preserving carrier collection.
\end{itemize}
\end{column}
\begin{column}{0.4\textwidth}
\centering
\begin{tabular}{lr}
\toprule
Layer & Thickness \\
\midrule
n++ Ge & \SI{50}{\nano\metre} \\
i-Ge & \SI{350}{\nano\metre} \\
p++ Si & \SI{220}{\nano\metre} \\
BOX & \SI{2}{\micro\metre} \\
\bottomrule
\end{tabular}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Optical Simulation Flow}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}
\item \textbf{Lumerical FDTD} models the taper-coupled optical absorption at \SI{1310}{\nano\metre}.
\item The simulation exports \textbf{optical generation rate} into CHARGE.
\item Ge length is optimized to reach high absorption with minimum footprint.
\end{itemize}
\vspace{0.5em}
\textbf{Key optical design points:}
\begin{itemize}
\item TE-driven O-band operation
\item Adiabatic spot-size converter (\SI{500}{\nano\metre} $\to$ \SI{5}{\micro\metre} over \SI{40}{\micro\metre})
\item 3D generation-rate export: $G = P_\mathrm{abs}/(h\nu)$
\end{itemize}
\end{column}

\begin{frame}{Electrical Simulation Flow}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}
\item \textbf{Lumerical DEVICE CHARGE} solves carrier transport under reverse bias.
\item SRH recombination enabled with the implemented Ge lifetime $\tau_n=\tau_p=\SI{1}{\nano\second}$.
\item Fine mesh override: \SI{20}{\nano\metre} max edge length over the Ge active region.
\item Extracted quantities include:
\begin{itemize}
\item Dark current and illuminated current
\item Band diagram, carrier density
\item Electric field and potential
\end{itemize}
\item Bandwidth estimated from \textbf{transit-time + RC} limits.
\end{itemize}
\end{column}

\vspace{0.5em}
\normalsize
\begin{block}{}
All parameters flow from Lumerical solvers $\to$ MAT files $\to$ MATLAB postprocess $\to$ CML export $\to$ system-level scripts. \textbf{Nothing is hardcoded in the pipeline.}
\end{block}
\end{frame}

\begin{frame}{Final Takeaways}
\begin{itemize}
\item The chapter demonstrates a \textbf{Ge-on-Si O-band photodiode} targeting published high-speed metrics.
\item The workflow is consistent across:
\begin{itemize}
        \item \textbf{Device-level FDTD}---optical absorption, generation rate
        \item \textbf{Device-level CHARGE}---dark current, responsivity, band structure
        \item \textbf{Device-level postprocess}---CML dataset, all derived parameters
        \item \textbf{System-level MATLAB}---PAM-4 BER, eye diagrams, histograms
        \item \textbf{INTERCONNECT-ready export}---circuit-level integration
\end{itemize}
\item \textbf{Zero hardcoded paper values}---all metrics from solvers and geometry.
\end{itemize}
\end{frame}
